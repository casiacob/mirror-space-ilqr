% Shared pgfplots setup of the figures. Load it in the preamble: \input{style}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.16}

% Directory of the csv files, relative to the directory pdflatex runs in.
\providecommand{\datadir}{data}

\pgfplotsset{
  paper/.style={
    scale only axis,
    width=3.3cm,
    height=2.6cm,
    enlargelimits=0.05,
    minor x tick num=1,
    xmajorgrids,
    xminorgrids,
    ymajorgrids,
    grid style={gray, line width=0.3pt},
    axis line style={line width=0.5pt},
    legend cell align=left,
    legend style={font=\scriptsize, row sep=-2pt},
    table/col sep=comma,
    cycle list={{black, solid}, {black, dashed}},
    every axis plot/.append style={line width=1pt, no markers},
  },
}
